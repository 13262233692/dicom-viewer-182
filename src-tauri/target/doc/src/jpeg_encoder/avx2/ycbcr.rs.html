17macro_rules! ycbcr_image_avx2 {
18    ($name:ident, $num_colors:expr, $o1:expr, $o2:expr, $o3:expr) => {
19        pub(crate) struct $name<'a>(pub &'a [u8], pub u16, pub u16);
20
21        impl<'a> $name<'a> {
22            #[target_feature(enable = "avx2")]
23            unsafe fn fill_buffers_avx2(&self, y: u16, buffers: &mut [Vec<u8>; 4]) {
24                unsafe fn load3(data: *const u8) -> __m256i {
25                    _mm256_set_epi32(
26                        *data as i32,
27                        *data.offset(1 * $num_colors) as i32,
28                        *data.offset(2 * $num_colors) as i32,
29                        *data.offset(3 * $num_colors) as i32,
30                        *data.offset(4 * $num_colors) as i32,
31                        *data.offset(5 * $num_colors) as i32,
32                        *data.offset(6 * $num_colors) as i32,
33                        *data.offset(7 * $num_colors) as i32,
34                    )
35                }
36
37                let mut y_buffer = buffers[0].as_mut_ptr().add(buffers[0].len());
38                buffers[0].set_len(buffers[0].len() + self.width() as usize);
39                let mut cb_buffer = buffers[1].as_mut_ptr().add(buffers[1].len());
40                buffers[1].set_len(buffers[1].len() + self.width() as usize);
41                let mut cr_buffer = buffers[2].as_mut_ptr().add(buffers[2].len());
42                buffers[2].set_len(buffers[2].len() + self.width() as usize);
43
44                let ymulr = _mm256_set1_epi32(19595);
45                let ymulg = _mm256_set1_epi32(38470);
46                let ymulb = _mm256_set1_epi32(7471);
47
48                let cbmulr = _mm256_set1_epi32(-11059);
49                let cbmulg = _mm256_set1_epi32(21709);
50                let cbmulb = _mm256_set1_epi32(32768);
51
52                let crmulr = _mm256_set1_epi32(32768);
53                let crmulg = _mm256_set1_epi32(27439);
54                let crmulb = _mm256_set1_epi32(5329);
55
56                let mut data = self
57                    .0
58                    .as_ptr()
59                    .offset((y as isize * self.1 as isize * $num_colors));
60
61                for _ in 0..self.width() / 8 {
62                    let r = load3(data.offset($o1));
63                    let g = load3(data.offset($o2));
64                    let b = load3(data.offset($o3));
65
66                    data = data.add($num_colors * 8);
67
68                    let yr = _mm256_mullo_epi32(ymulr, r);
69                    let yg = _mm256_mullo_epi32(ymulg, g);
70                    let yb = _mm256_mullo_epi32(ymulb, b);
71
72                    let y = _mm256_add_epi32(_mm256_add_epi32(yr, yg), yb);
73                    let y = _mm256_add_epi32(y, _mm256_set1_epi32(0x7FFF));
74                    let y = _mm256_srli_epi32(y, 16);
75                    let y: [i32; 8] = core::mem::transmute(y);
76
77                    let cbr = _mm256_mullo_epi32(cbmulr, r);
78                    let cbg = _mm256_mullo_epi32(cbmulg, g);
79                    let cbb = _mm256_mullo_epi32(cbmulb, b);
80
81                    let cb = _mm256_add_epi32(_mm256_sub_epi32(cbr, cbg), cbb);
82                    let cb = _mm256_add_epi32(cb, _mm256_set1_epi32(128 << 16));
83                    let cb = _mm256_add_epi32(cb, _mm256_set1_epi32(0x7FFF));
84                    let cb = _mm256_srli_epi32(cb, 16);
85                    let cb: [i32; 8] = core::mem::transmute(cb);
86
87                    let crr = _mm256_mullo_epi32(crmulr, r);
88                    let crg = _mm256_mullo_epi32(crmulg, g);
89                    let crb = _mm256_mullo_epi32(crmulb, b);
90
91                    let cr = _mm256_sub_epi32(_mm256_sub_epi32(crr, crg), crb);
92                    let cr = _mm256_add_epi32(cr, _mm256_set1_epi32(128 << 16));
93                    let cr = _mm256_add_epi32(cr, _mm256_set1_epi32(0x7FFF));
94                    let cr = _mm256_srli_epi32(cr, 16);
95                    let cr: [i32; 8] = core::mem::transmute(cr);
96
97                    for y in y.iter().rev() {
98                        *y_buffer = *y as u8;
99                        y_buffer = y_buffer.offset(1);
100                    }
101
102                    for cb in cb.iter().rev() {
103                        *cb_buffer = *cb as u8;
104                        cb_buffer = cb_buffer.offset(1);
105                    }
106
107                    for cr in cr.iter().rev() {
108                        *cr_buffer = *cr as u8;
109                        cr_buffer = cr_buffer.offset(1);
110                    }
111                }
112
113                for _ in 0..self.width() % 8 {
114                    let (y, cb, cr) =
115                        rgb_to_ycbcr(*data.offset($o1), *data.offset($o2), *data.offset($o3));
116
117                    data = data.add($num_colors);
118
119                    *y_buffer = y;
120                    y_buffer = y_buffer.offset(1);
121
122                    *cb_buffer = cb;
123                    cb_buffer = cb_buffer.offset(1);
124
125                    *cr_buffer = cr;
126                    cr_buffer = cr_buffer.offset(1);
127                }
128            }
129        }
130
131        impl<'a> ImageBuffer for $name<'a> {
132            fn get_jpeg_color_type(&self) -> JpegColorType {
133                JpegColorType::Ycbcr
134            }
135
136            fn width(&self) -> u16 {
137                self.1
138            }
139
140            fn height(&self) -> u16 {
141                self.2
142            }
143
144            #[inline(always)]
145            fn fill_buffers(&self, y: u16, buffers: &mut [Vec<u8>; 4]) {
146                unsafe {
147                    self.fill_buffers_avx2(y, buffers);
148                }
149            }
150        }
151    };
152}
