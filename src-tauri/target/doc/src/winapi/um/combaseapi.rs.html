24#[inline]
25pub fn LISet32(li: &mut LARGE_INTEGER, v: DWORD) {
26    unsafe {
27        li.u_mut().HighPart = if (v as LONG) < 0 {
28            -1
29        } else {
30            0
31        };
32        li.u_mut().LowPart = v;
33    }
34}
35#[inline]
36pub fn ULISet32(li: &mut ULARGE_INTEGER, v: DWORD) {
37    unsafe {
38        li.u_mut().HighPart = 0;
39        li.u_mut().LowPart = v;
40    }
41}
