1//! Libc s(n)printf clone written in Rust, so you can use printf-style
2//! formatting without a libc (e.g. in WebAssembly).
3//!
4//! **Note:** *You're probably better off using standard Rust string formatting
5//! instead of thie crate unless you specificaly need printf compatibility.*
6//!
7//! It follows the standard C semantics, except:
8//!
9//!  * Locale-aware UNIX extensions (`'` and GNU’s `I`) are not supported.
10//!  * `%a`/`%A` (hexadecimal floating point) are currently not implemented.
11//!  * Length modifiers (`h`, `l`, etc.) are checked, but ignored. The passed
12//!    type is used instead.
13//!
14//! Usage example:
15//!
16//!     use sprintf::sprintf;
17//!     let s = sprintf!("%d + %d = %d\n", 3, 9, 3+9).unwrap();
18//!     assert_eq!(s, "3 + 9 = 12\n");
19//!
20//! The types of the arguments are checked at runtime.
21//!

30/// Error type
31#[derive(Debug, Clone, Copy)]
32pub enum PrintfError {
33    /// Error parsing the format string
34    ParseError,
35    /// Incorrect type passed as an argument
36    WrongType,
37    /// Too many arguments passed
38    TooManyArgs,
39    /// Too few arguments passed
40    NotEnoughArgs,
41    /// Other error (should never happen)
42    Unknown,
43}
44
45pub type Result<T> = std::result::Result<T, PrintfError>;

47/// Format a string. (Roughly equivalent to `vsnprintf` or `vasprintf` in C)
48///
49/// Takes a printf-style format string `format` and a slice of dynamically
50/// typed arguments, `args`.
51///
52///     use sprintf::{vsprintf, Printf};
53///     let n = 16;
54///     let args: Vec<&dyn Printf> = vec![&n];
55///     let s = vsprintf("%#06x", &args).unwrap();
56///     assert_eq!(s, "0x0010");
57///
58/// See also: [sprintf]
59pub fn vsprintf(format: &str, args: &[&dyn Printf]) -> Result<String> {
60    vsprintfp(&parse_format_string(format)?, args)
61}
62
63fn vsprintfp(format: &[FormatElement], args: &[&dyn Printf]) -> Result<String> {
64    let mut res = String::new();
65
66    let mut args = args;
67    let mut pop_arg = || {
68        if args.is_empty() {
69            Err(PrintfError::NotEnoughArgs)
70        } else {
71            let a = args[0];
72            args = &args[1..];
73            Ok(a)
74        }
75    };
76
77    for elem in format {
78        match elem {
79            FormatElement::Verbatim(s) => {
80                res.push_str(s);
81            }
82            FormatElement::Format(spec) => {
83                if spec.conversion_type == ConversionType::PercentSign {
84                    res.push('%');
85                } else {
86                    let mut completed_spec = *spec;
87                    if spec.width == NumericParam::FromArgument {
88                        completed_spec.width = NumericParam::Literal(
89                            pop_arg()?.as_int().ok_or(PrintfError::WrongType)?,
90                        )
91                    }
92                    if spec.precision == NumericParam::FromArgument {
93                        completed_spec.precision = NumericParam::Literal(
94                            pop_arg()?.as_int().ok_or(PrintfError::WrongType)?,
95                        )
96                    }
97                    res.push_str(&pop_arg()?.format(&completed_spec)?);
98                }
99            }
100        }
101    }
102
103    if args.is_empty() {
104        Ok(res)
105    } else {
106        Err(PrintfError::TooManyArgs)
107    }
108}

110/// Format a string. (Roughly equivalent to `snprintf` or `asprintf` in C)
111///
112/// Takes a printf-style format string `format` and a variable number of
113/// additional arguments.
114///
115///     use sprintf::sprintf;
116///     let s = sprintf!("%s = %*d", "forty-two", 4, 42).unwrap();
117///     assert_eq!(s, "forty-two =   42");
118///
119/// Wrapper around [vsprintf].
120#[macro_export]
121macro_rules! sprintf {
122    ($fmt:expr, $($arg:expr),*) => {
123        sprintf::vsprintf($fmt, &[$( &($arg) as &dyn sprintf::Printf),* ][..])
124    };
125}
