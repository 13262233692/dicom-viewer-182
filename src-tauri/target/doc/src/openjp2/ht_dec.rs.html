168/* * @brief Reads and unstuffs the MEL bitstream
169 *
170 *  This design needs more bytes in the codeblock buffer than the length
171 *  of the cleanup pass by up to 2 bytes.
172 *
173 *  Unstuffing removes the MSB of the byte following a byte whose
174 *  value is 0xFF; this prevents sequences larger than 0xFF7F in value
175 *  from appearing the bitstream.
176 *
177 *  @param [in]  melp is a pointer to dec_mel_t structure
178 */
179#[inline]
180unsafe fn mel_read(mut melp: *mut dec_mel_t) {
181  let mut val: OPJ_UINT32 = 0;
182  let mut bits: core::ffi::c_int = 0;
183  let mut t: OPJ_UINT32 = 0;
184  let mut unstuff: OPJ_BOOL = 0;
185  if (*melp).bits > 32i32 {
186    //there are enough bits in the tmp variable
187    return;
188    // return without reading new data
189  } // feed in 0xFF if buffer is exhausted
190  val = 0xffffffffu32;
191  if (*melp).size > 4i32 {
192    // if there is more than 4 bytes the MEL segment
193    val = read_le_uint32((*melp).data as *const core::ffi::c_void); // read 32 bits from MEL data
194                                                                    // reduce counter
195    (*melp).data = (*melp).data.offset(4); // advance pointer
196    (*melp).size -= 4i32
197  } else if (*melp).size > 0i32 {
198    // 4 or less
199    let mut m: OPJ_UINT32 = 0; // read one byte at a time
200    let mut v: OPJ_UINT32 = 0; // mask of location
201    let mut i = 0i32; // put byte in its correct location
202    while (*melp).size > 1i32 {
203      let fresh0 = (*melp).data;
204      (*melp).data = (*melp).data.offset(1);
205      let mut v_0 = *fresh0 as OPJ_UINT32;
206      let mut m_0 = !((0xffu32) << i);
207      val = val & m_0 | v_0 << i;
208      (*melp).size -= 1;
209      i += 8i32
210    }
211    // size equal to 1
212    let fresh1 = (*melp).data; // the one before the last is different
213    (*melp).data = (*melp).data.offset(1); // MEL and VLC segments can overlap
214    v = *fresh1 as OPJ_UINT32;
215    v |= 0xfu32;
216    m = !((0xffu32) << i);
217    val = val & m | v << i;
218    (*melp).size -= 1
219  }
220  // next we unstuff them before adding them to the buffer
221  bits = 32i32 - (*melp).unstuff; // number of bits in val, subtract 1 if
222                                  // the previously read byte requires
223                                  // unstuffing
224                                  // data is unstuffed and accumulated in t
225                                  // bits has the number of bits in t
226  t = val & 0xffu32; // true if the byte needs unstuffing
227  unstuff = (val & 0xffu32 == 0xffu32) as core::ffi::c_int; // there is one less bit in t if unstuffing is needed
228  bits -= unstuff; // move up to make room for the next byte
229  t <<= 8i32 - unstuff;
230  //this is a repeat of the above
231  t |= val >> 8i32 & 0xffu32;
232  unstuff = (val >> 8i32 & 0xffu32 == 0xffu32) as core::ffi::c_int;
233  bits -= unstuff;
234  t <<= 8i32 - unstuff;
235  t |= val >> 16i32 & 0xffu32;
236  unstuff = (val >> 16i32 & 0xffu32 == 0xffu32) as core::ffi::c_int;
237  bits -= unstuff;
238  t <<= 8i32 - unstuff;
239  t |= val >> 24i32 & 0xffu32;
240  (*melp).unstuff = (val >> 24i32 & 0xffu32 == 0xffu32) as core::ffi::c_int;
241  // move t to tmp, and push the result all the way up, so we read from
242  // the MSB
243  (*melp).tmp |= (t as OPJ_UINT64) << (64i32 - bits - (*melp).bits);
244  (*melp).bits += bits;
245  //increment the number of bits in tmp
246}

248/* * @brief Decodes unstuffed MEL segment bits stored in tmp to runs
249 *
250 *  Runs are stored in "runs" and the number of runs in "num_runs".
251 *  Each run represents a number of zero events that may or may not
252 *  terminate in a 1 event.
253 *  Each run is stored in 7 bits.  The LSB is 1 if the run terminates in
254 *  a 1 event, 0 otherwise.  The next 6 bits, for the case terminating
255 *  with 1, contain the number of consecutive 0 zero events * 2; for the
256 *  case terminating with 0, they store (number of consecutive 0 zero
257 *  events - 1) * 2.
258 *  A total of 6 bits (made up of 1 + 5) should have been enough.
259 *
260 *  @param [in]  melp is a pointer to dec_mel_t structure
261 */
262#[inline]
263unsafe fn mel_decode(mut melp: *mut dec_mel_t) {
264  const mel_exp: [core::ffi::c_int; 13] = [0, 0, 0, 1, 1, 1, 2, 2, 2, 3, 3, 4, 5];
265  if (*melp).bits < 6i32 {
266    // if there are less than 6 bits in tmp
267    mel_read(melp);
268    // then read from the MEL bitstream
269  }
270  // 6 bits is the largest decodable MEL cwd
271  //repeat so long that there is enough decodable bits in tmp,
272  // and the runs store is not full (num_runs < 8)
273  while (*melp).bits >= 6i32 && (*melp).num_runs < 8i32 {
274    let mut eval = mel_exp[(*melp).k as usize]; // number of bits associated with state
275    let mut run = 0i32;
276    if (*melp).tmp & (1u64) << 63i32 != 0 {
277      //The next bit to decode (stored in MSB)
278      //one is found
279      run = (1i32) << eval;
280      // a stretch of zeros not terminating in one
281      run -= 1; // consecutive runs of 0 events - 1
282      (*melp).k = if ((*melp).k + 1i32) < 12i32 {
283        ((*melp).k) + 1i32
284      } else {
285        12i32
286      }; //increment, max is 12
287      (*melp).tmp <<= 1i32; // consume one bit from tmp
288      (*melp).bits -= 1i32;
289      run <<= 1i32
290    } else {
291      //0 is found
292      run = ((*melp).tmp >> (63i32 - eval)) as core::ffi::c_int & (((1i32) << eval) - 1i32);
293      // a stretch of zeros terminating with one
294      (*melp).k = if (*melp).k - 1i32 > 0i32 {
295        ((*melp).k) - 1i32
296      } else {
297        0i32
298      }; //decrement, min is 0
299      (*melp).tmp <<= eval + 1i32; //consume eval + 1 bits (max is 6)
300      (*melp).bits -= eval + 1i32; // 7 bits per run
301      run = (run << 1i32) + 1i32
302    } // 6 bits are sufficient
303    eval = (*melp).num_runs * 7i32; // store the value in runs
304    (*melp).runs &= !((0x3f as OPJ_UINT64) << eval);
305    (*melp).runs |= (run as OPJ_UINT64) << eval;
306    (*melp).num_runs += 1
307  }
308}

310/* * @brief Initiates a dec_mel_t structure for MEL decoding and reads
311 *         some bytes in order to get the read address to a multiple
312 *         of 4
313 *
314 *  @param [in]  melp is a pointer to dec_mel_t structure
315 *  @param [in]  bbuf is a pointer to byte buffer
316 *  @param [in]  lcup is the length of MagSgn+MEL+VLC segments
317 *  @param [in]  scup is the length of MEL+VLC segments
318 */
319#[inline]
320unsafe fn mel_init(
321  mut melp: *mut dec_mel_t,
322  mut bbuf: *mut OPJ_UINT8,
323  mut lcup: core::ffi::c_int,
324  mut scup: core::ffi::c_int,
325) -> bool {
326  let mut num: core::ffi::c_int = 0; // move the pointer to the start of MEL
327  let mut i: core::ffi::c_int = 0; // 0 bits in tmp
328  (*melp).data = bbuf.offset(lcup as isize).offset(-(scup as isize)); //
329  (*melp).bits = 0i32; // no unstuffing
330  (*melp).tmp = 0 as OPJ_UINT64; // size is the length of MEL+VLC-1
331  (*melp).unstuff = 0i32; // 0 for state
332  (*melp).size = scup - 1i32; // num_runs is 0
333  (*melp).k = 0i32; //
334  (*melp).num_runs = 0i32;
335  (*melp).runs = 0 as OPJ_UINT64;
336  //This code is borrowed; original is for a different architecture
337  //These few lines take care of the case where data is not at a multiple
338  // of 4 boundary.  It reads 1,2,3 up to 4 bytes from the MEL segment
339  num = 4i32 - ((*melp).data as usize & 0x3) as core::ffi::c_int;
340  i = 0i32;
341  while i < num {
342    // this code is similar to mel_read
343    let mut d: OPJ_UINT64 = 0; // if buffer is consumed
344    let mut d_bits: core::ffi::c_int = 0;
345    if (*melp).unstuff != 0i32 && *(*melp).data.offset(0) as core::ffi::c_int > 0x8fi32 {
346      return false;
347    }
348    d = if (*melp).size > 0i32 {
349      *(*melp).data as core::ffi::c_int
350    } else {
351      0xffi32
352    } as OPJ_UINT64;
353    // set data to 0xFF
354    if (*melp).size == 1i32 {
355      d |= 0xfu64
356      //if this is MEL+VLC-1, set LSBs to 0xF
357    }
358    // see the standard
359    let fresh2 = (*melp).size; //increment if the end is not reached
360    (*melp).size -= 1; //if unstuffing is needed, reduce by 1
361    (*melp).data = (*melp).data.offset((fresh2 > 0i32) as isize); //store bits in tmp
362    d_bits = 8i32 - (*melp).unstuff; //increment tmp by number of bits
363    (*melp).tmp = (*melp).tmp << d_bits | d;
364    (*melp).bits += d_bits;
365    (*melp).unstuff = (d & 0xffu64 == 0xffu64) as core::ffi::c_int;
366    i += 1
367  }
368  (*melp).tmp <<= 64i32 - (*melp).bits;
369  //push all the way up so the first bit
370  // is the MSB
371  true
372}

393/* * @brief Read and unstuff data from a backwardly-growing segment
394 *
395 *  This reader can read up to 8 bytes from before the VLC segment.
396 *  Care must be taken not read from unreadable memory, causing a
397 *  segmentation fault.
398 *
399 *  Note that there is another subroutine rev_read_mrp that is slightly
400 *  different.  The other one fills zeros when the buffer is exhausted.
401 *  This one basically does not care if the bytes are consumed, because
402 *  any extra data should not be used in the actual decoding.
403 *
404 *  Unstuffing is needed to prevent sequences more than 0xFF8F from
405 *  appearing in the bits stream; since we are reading backward, we keep
406 *  watch when a value larger than 0x8F appears in the bitstream.
407 *  If the byte following this is 0x7F, we unstuff this byte (ignore the
408 *  MSB of that byte, which should be 0).
409 *
410 *  @param [in]  vlcp is a pointer to rev_struct_t structure
411 */
412#[inline]
413unsafe fn rev_read(mut vlcp: *mut rev_struct_t) {
414  let mut val: OPJ_UINT32 = 0;
415  let mut tmp: OPJ_UINT32 = 0;
416  let mut bits: OPJ_UINT32 = 0;
417  let mut unstuff: OPJ_BOOL = 0;
418  //process 4 bytes at a time
419  if (*vlcp).bits > 32u32 {
420    // if there are more than 32 bits in tmp, then
421    return;
422    // reading 32 bits can overflow vlcp->tmp
423  }
424  val = 0 as OPJ_UINT32;
425  //the next line (the if statement) needs to be tested first
426  if (*vlcp).size > 3i32 {
427    // if there are more than 3 bytes left in VLC
428    // (vlcp->data - 3) move pointer back to read 32 bits at once
429    val = read_le_uint32((*vlcp).data.offset(-3) as *const core::ffi::c_void); // then read 32 bits
430                                                                               // reduce available byte by 4
431    (*vlcp).data = (*vlcp).data.offset(-4); // move data pointer back by 4
432    (*vlcp).size -= 4i32
433  } else if (*vlcp).size > 0i32 {
434    // 4 or less
435    let mut i = 24i32; // read one byte at a time
436    while (*vlcp).size > 0i32 {
437      let fresh3 = (*vlcp).data; // put byte in its correct location
438      (*vlcp).data = (*vlcp).data.offset(-1);
439      let mut v = *fresh3 as OPJ_UINT32;
440      val |= v << i;
441      (*vlcp).size -= 1;
442      i -= 8i32
443    }
444  }
445  //accumulate in tmp, number of bits in tmp are stored in bits
446  tmp = val >> 24i32; //start with the MSB byte
447                      // test unstuff (previous byte is >0x8F), and this byte is 0x7F
448  bits = (8u32).wrapping_sub(
449    if (*vlcp).unstuff != 0 && val >> 24i32 & 0x7fu32 == 0x7fu32 {
450      1u32
451    } else {
452      0u32
453    },
454  ); //this is for the next byte
455  unstuff = (val >> 24i32 > 0x8fu32) as core::ffi::c_int; //process the next byte
456  tmp |= (val >> 16i32 & 0xffu32) << bits;
457  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(
458    if unstuff != 0 && val >> 16i32 & 0x7fu32 == 0x7fu32 {
459      1u32
460    } else {
461      0u32
462    },
463  )) as OPJ_UINT32;
464  unstuff = (val >> 16i32 & 0xffu32 > 0x8fu32) as core::ffi::c_int;
465  tmp |= (val >> 8i32 & 0xffu32) << bits;
466  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(
467    if unstuff != 0 && val >> 8i32 & 0x7fu32 == 0x7fu32 {
468      1u32
469    } else {
470      0u32
471    },
472  )) as OPJ_UINT32;
473  unstuff = (val >> 8i32 & 0xffu32 > 0x8fu32) as core::ffi::c_int;
474  tmp |= (val & 0xffu32) << bits;
475  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(
476    if unstuff != 0 && val & 0x7fu32 == 0x7fu32 {
477      1u32
478    } else {
479      0u32
480    },
481  )) as OPJ_UINT32;
482  unstuff = (val & 0xffu32 > 0x8fu32) as core::ffi::c_int;
483  // now move the read and unstuffed bits into vlcp->tmp
484  (*vlcp).tmp |= (tmp as OPJ_UINT64) << (*vlcp).bits;
485  (*vlcp).bits = ((*vlcp).bits as core::ffi::c_uint).wrapping_add(bits) as OPJ_UINT32;
486  (*vlcp).unstuff = unstuff;
487  // this for the next read
488}
489//* ***********************************************************************/
490/* * @brief Initiates the rev_struct_t structure and reads a few bytes to
491 *         move the read address to multiple of 4
492 *
493 *  There is another similar rev_init_mrp subroutine.  The difference is
494 *  that this one, rev_init, discards the first 12 bits (they have the
495 *  sum of the lengths of VLC and MEL segments), and first unstuff depends
496 *  on first 4 bits.
497 *
498 *  @param [in]  vlcp is a pointer to rev_struct_t structure
499 *  @param [in]  data is a pointer to byte at the start of the cleanup pass
500 *  @param [in]  lcup is the length of MagSgn+MEL+VLC segments
501 *  @param [in]  scup is the length of MEL+VLC segments
502 */
503#[inline]
504unsafe fn rev_init(
505  mut vlcp: *mut rev_struct_t,
506  mut data: *mut OPJ_UINT8,
507  mut lcup: core::ffi::c_int,
508  mut scup: core::ffi::c_int,
509) {
510  let mut d: OPJ_UINT32 = 0;
511  let mut num: core::ffi::c_int = 0;
512  let mut tnum: core::ffi::c_int = 0;
513  let mut i: core::ffi::c_int = 0;
514  //first byte has only the upper 4 bits
515  (*vlcp).data = data.offset(lcup as isize).offset(-2);
516  //size can not be larger than this, in fact it should be smaller
517  (*vlcp).size = scup - 2i32; // read one byte (this is a half byte)
518  let fresh4 = (*vlcp).data; // both initialize and set
519  (*vlcp).data = (*vlcp).data.offset(-1); //check standard
520  d = *fresh4 as OPJ_UINT32; //this is useful for the next byte
521  (*vlcp).tmp = (d >> 4i32) as OPJ_UINT64;
522  (*vlcp).bits = (4i32 - ((*vlcp).tmp & 7u64 == 7u64) as core::ffi::c_int) as OPJ_UINT32;
523  (*vlcp).unstuff = (d | 0xfu32 > 0x8fu32) as core::ffi::c_int;
524  //This code is designed for an architecture that read address should
525  // align to the read size (address multiple of 4 if read size is 4)
526  //These few lines take care of the case where data is not at a multiple
527  // of 4 boundary. It reads 1,2,3 up to 4 bytes from the VLC bitstream.
528  // To read 32 bits, read from (vlcp->data - 3)
529  num = 1i32 + ((*vlcp).data as usize & 0x3) as core::ffi::c_int;
530  tnum = if num < (*vlcp).size {
531    num
532  } else {
533    (*vlcp).size
534  };
535  i = 0i32;
536  while i < tnum {
537    let mut d_0: OPJ_UINT64 = 0;
538    let mut d_bits: OPJ_UINT32 = 0;
539    // for next byte
540    let fresh5 = (*vlcp).data; // read one byte and move read pointer
541    (*vlcp).data = (*vlcp).data.offset(-1);
542    d_0 = *fresh5 as OPJ_UINT64;
543    d_bits = (8u32).wrapping_sub(if (*vlcp).unstuff != 0 && d_0 & 0x7fu64 == 0x7fu64 {
544      1u32
545    } else {
546      0u32
547    });
548    (*vlcp).tmp |= d_0 << (*vlcp).bits;
549    (*vlcp).bits = ((*vlcp).bits as core::ffi::c_uint).wrapping_add(d_bits) as OPJ_UINT32;
550    (*vlcp).unstuff = (d_0 > 0x8fu64) as core::ffi::c_int;
551    i += 1
552  }
553  (*vlcp).size -= tnum;
554  rev_read(vlcp);
555  //check if the last byte was >0x8F (unstuff == true) and this is 0x7F
556  // move data to vlcp->tmp
557  // read another 32 buts
558}
559//* ***********************************************************************/
560/* * @brief Retrieves 32 bits from the head of a rev_struct structure
561 *
562 *  By the end of this call, vlcp->tmp must have no less than 33 bits
563 *
564 *  @param [in]  vlcp is a pointer to rev_struct structure
565 */
566#[inline]
567unsafe fn rev_fetch(mut vlcp: *mut rev_struct_t) -> OPJ_UINT32 {
568  if (*vlcp).bits < 32u32 {
569    // if there are less then 32 bits, read more
570    rev_read(vlcp); // read 32 bits, but unstuffing might reduce this
571    if (*vlcp).bits < 32u32 {
572      // if there is still space in vlcp->tmp for 32 bits
573      rev_read(vlcp);
574      // read another 32
575    }
576  }
577  (*vlcp).tmp as OPJ_UINT32
578  // return the head (bottom-most) of vlcp->tmp
579}

581/* * @brief Consumes num_bits from a rev_struct structure
582 *
583 *  @param [in]  vlcp is a pointer to rev_struct structure
584 *  @param [in]  num_bits is the number of bits to be removed
585 */
586#[inline]
587unsafe fn rev_advance(mut vlcp: *mut rev_struct_t, mut num_bits: OPJ_UINT32) -> OPJ_UINT32 {
588  assert!(num_bits <= (*vlcp).bits); // remove bits
589  (*vlcp).tmp >>= num_bits; // decrement the number of bits
590  (*vlcp).bits = ((*vlcp).bits as core::ffi::c_uint).wrapping_sub(num_bits) as OPJ_UINT32;
591  (*vlcp).tmp as OPJ_UINT32
592}

594/* * @brief Reads and unstuffs from rev_struct
595 *
596 *  This is different than rev_read in that this fills in zeros when the
597 *  the available data is consumed.  The other does not care about the
598 *  values when all data is consumed.
599 *
600 *  See rev_read for more information about unstuffing
601 *
602 *  @param [in]  mrp is a pointer to rev_struct structure
603 */
604#[inline]
605unsafe fn rev_read_mrp(mut mrp: *mut rev_struct_t) {
606  let mut val: OPJ_UINT32 = 0;
607  let mut tmp: OPJ_UINT32 = 0;
608  let mut bits: OPJ_UINT32 = 0;
609  let mut unstuff: OPJ_BOOL = 0;
610  //process 4 bytes at a time
611  if (*mrp).bits > 32u32 {
612    return;
613  }
614  val = 0 as OPJ_UINT32;
615  if (*mrp).size > 3i32 {
616    // If there are 3 byte or more
617    // (mrp->data - 3) move pointer back to read 32 bits at once
618    val = read_le_uint32((*mrp).data.offset(-3) as *const core::ffi::c_void); // read 32 bits
619                                                                              // reduce count
620    (*mrp).data = (*mrp).data.offset(-4); // move back pointer
621    (*mrp).size -= 4i32
622  } else if (*mrp).size > 0i32 {
623    let mut i = 24i32; // read one byte at a time
624    while (*mrp).size > 0i32 {
625      let fresh6 = (*mrp).data; // put byte in its correct location
626      (*mrp).data = (*mrp).data.offset(-1);
627      let mut v = *fresh6 as OPJ_UINT32;
628      val |= v << i;
629      (*mrp).size -= 1;
630      i -= 8i32
631    }
632  }
633  //accumulate in tmp, and keep count in bits
634  tmp = val >> 24i32;
635  //test if the last byte > 0x8F (unstuff must be true) and this is 0x7F
636  bits = (8u32).wrapping_sub(
637    if (*mrp).unstuff != 0 && val >> 24i32 & 0x7fu32 == 0x7fu32 {
638      1u32
639    } else {
640      0u32
641    },
642  );
643  unstuff = (val >> 24i32 > 0x8fu32) as core::ffi::c_int;
644  //process the next byte
645  tmp |= (val >> 16i32 & 0xffu32) << bits; // move data to mrp pointer
646  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(
647    if unstuff != 0 && val >> 16i32 & 0x7fu32 == 0x7fu32 {
648      1u32
649    } else {
650      0u32
651    },
652  )) as OPJ_UINT32;
653  unstuff = (val >> 16i32 & 0xffu32 > 0x8fu32) as core::ffi::c_int;
654  tmp |= (val >> 8i32 & 0xffu32) << bits;
655  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(
656    if unstuff != 0 && val >> 8i32 & 0x7fu32 == 0x7fu32 {
657      1u32
658    } else {
659      0u32
660    },
661  )) as OPJ_UINT32;
662  unstuff = (val >> 8i32 & 0xffu32 > 0x8fu32) as core::ffi::c_int;
663  tmp |= (val & 0xffu32) << bits;
664  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(
665    if unstuff != 0 && val & 0x7fu32 == 0x7fu32 {
666      1u32
667    } else {
668      0u32
669    },
670  )) as OPJ_UINT32;
671  unstuff = (val & 0xffu32 > 0x8fu32) as core::ffi::c_int;
672  (*mrp).tmp |= (tmp as OPJ_UINT64) << (*mrp).bits;
673  (*mrp).bits = ((*mrp).bits as core::ffi::c_uint).wrapping_add(bits) as OPJ_UINT32;
674  (*mrp).unstuff = unstuff;
675  // next byte
676}

678/* * @brief Initialized rev_struct structure for MRP segment, and reads
679 *         a number of bytes such that the next 32 bits read are from
680 *         an address that is a multiple of 4. Note this is designed for
681 *         an architecture that read size must be compatible with the
682 *         alignment of the read address
683 *
684 *  There is another similar subroutine rev_init.  This subroutine does
685 *  NOT skip the first 12 bits, and starts with unstuff set to true.
686 *
687 *  @param [in]  mrp is a pointer to rev_struct structure
688 *  @param [in]  data is a pointer to byte at the start of the cleanup pass
689 *  @param [in]  lcup is the length of MagSgn+MEL+VLC segments
690 *  @param [in]  len2 is the length of SPP+MRP segments
691 */
692#[inline]
693unsafe fn rev_init_mrp(
694  mut mrp: *mut rev_struct_t,
695  mut data: *mut OPJ_UINT8,
696  mut lcup: core::ffi::c_int,
697  mut len2: core::ffi::c_int,
698) {
699  let mut num: core::ffi::c_int = 0;
700  let mut i: core::ffi::c_int = 0;
701  (*mrp).data = data.offset(lcup as isize).offset(len2 as isize).offset(-1);
702  (*mrp).size = len2;
703  (*mrp).unstuff = 1i32;
704  (*mrp).bits = 0 as OPJ_UINT32;
705  (*mrp).tmp = 0 as OPJ_UINT64;
706  //This code is designed for an architecture that read address should
707  // align to the read size (address multiple of 4 if read size is 4)
708  //These few lines take care of the case where data is not at a multiple
709  // of 4 boundary.  It reads 1,2,3 up to 4 bytes from the MRP stream
710  num = 1i32 + ((*mrp).data as usize & 0x3) as core::ffi::c_int;
711  i = 0i32;
712  while i < num {
713    let mut d: OPJ_UINT64 = 0;
714    let mut d_bits: OPJ_UINT32 = 0;
715    // for next byte
716    let fresh7 = (*mrp).size;
717    (*mrp).size -= 1;
718    d = if fresh7 > 0i32 {
719      let fresh8 = (*mrp).data;
720      (*mrp).data = (*mrp).data.offset(-1);
721      *fresh8 as core::ffi::c_int
722    } else {
723      0i32
724    } as OPJ_UINT64;
725    d_bits = (8u32).wrapping_sub(if (*mrp).unstuff != 0 && d & 0x7fu64 == 0x7fu64 {
726      1u32
727    } else {
728      0u32
729    });
730    (*mrp).tmp |= d << (*mrp).bits;
731    (*mrp).bits = ((*mrp).bits as core::ffi::c_uint).wrapping_add(d_bits) as OPJ_UINT32;
732    (*mrp).unstuff = (d > 0x8fu64) as core::ffi::c_int;
733    i += 1
734  }
735  rev_read_mrp(mrp);
736}
737//read a byte, 0 if no more data
738//check if unstuffing is needed
739// move data to vlcp->tmp
740//* ***********************************************************************/
741/* * @brief Retrieves 32 bits from the head of a rev_struct structure
742 *
743 *  By the end of this call, mrp->tmp must have no less than 33 bits
744 *
745 *  @param [in]  mrp is a pointer to rev_struct structure
746 */
747#[inline]
748unsafe fn rev_fetch_mrp(mut mrp: *mut rev_struct_t) -> OPJ_UINT32 {
749  if (*mrp).bits < 32u32 {
750    // if there are less than 32 bits in mrp->tmp
751    rev_read_mrp(mrp); // read 30-32 bits from mrp
752    if (*mrp).bits < 32u32 {
753      // if there is a space of 32 bits
754      rev_read_mrp(mrp);
755      // read more
756    }
757  }
758  (*mrp).tmp as OPJ_UINT32
759  // return the head of mrp->tmp
760}

762/* * @brief Consumes num_bits from a rev_struct structure
763 *
764 *  @param [in]  mrp is a pointer to rev_struct structure
765 *  @param [in]  num_bits is the number of bits to be removed
766 */
767#[inline]
768unsafe fn rev_advance_mrp(mut mrp: *mut rev_struct_t, mut num_bits: OPJ_UINT32) -> OPJ_UINT32 {
769  assert!(num_bits <= (*mrp).bits); // discard the lowest num_bits bits
770  (*mrp).tmp >>= num_bits;
771  (*mrp).bits = ((*mrp).bits as core::ffi::c_uint).wrapping_sub(num_bits) as OPJ_UINT32;
772  (*mrp).tmp as OPJ_UINT32
773  // return data after consumption
774}

776/* * @brief Decode initial UVLC to get the u value (or u_q)
777 *
778 *  @param [in]  vlc is the head of the VLC bitstream
779 *  @param [in]  mode is 0, 1, 2, 3, or 4. Values in 0 to 3 are composed of
780 *               u_off of 1st quad and 2nd quad of a quad pair.  The value
781 *               4 occurs when both bits are 1, and the event decoded
782 *               from MEL bitstream is also 1.
783 *  @param [out] u is the u value (or u_q) + 1.  Note: we produce u + 1;
784 *               this value is a partial calculation of u + kappa.
785 */
786#[inline]
787unsafe fn decode_init_uvlc(
788  mut vlc: OPJ_UINT32,
789  mut mode: OPJ_UINT32,
790  mut u: *mut OPJ_UINT32,
791) -> OPJ_UINT32 {
792  //table stores possible decoding three bits from vlc
793  // there are 8 entries for xx1, x10, 100, 000, where x means do not care
794  // table value is made up of
795  // 2 bits in the LSB for prefix length
796  // 3 bits for suffix length
797  // 3 bits in the MSB for prefix value (u_pfx in Table 3 of ITU T.814)
798  static mut dec: [OPJ_UINT8; 8] = [
799    (3i32 | (5i32) << 2i32 | (5i32) << 5i32) as OPJ_UINT8,
800    (1i32 | (1i32) << 5i32) as OPJ_UINT8,
801    (2i32 | (2i32) << 5i32) as OPJ_UINT8,
802    (1i32 | (1i32) << 5i32) as OPJ_UINT8,
803    (3i32 | (1i32) << 2i32 | (3i32) << 5i32) as OPJ_UINT8,
804    (1i32 | (1i32) << 5i32) as OPJ_UINT8,
805    (2i32 | (2i32) << 5i32) as OPJ_UINT8,
806    (1i32 | (1i32) << 5i32) as OPJ_UINT8,
807  ];
808  let mut consumed_bits = 0 as OPJ_UINT32;
809  if mode == 0u32 {
810    // both u_off are 0
811    let fresh9 = &mut (*u.offset(1));
812    *fresh9 = 1 as OPJ_UINT32;
813    *u.offset(0) = *fresh9
814  //Kappa is 1 for initial line
815  } else if mode <= 2u32 {
816    // u_off are either 01 or 10
817    let mut d: OPJ_UINT32 = 0; //look at the least significant 3 bits
818    let mut suffix_len: OPJ_UINT32 = 0; //prefix length
819    d = dec[(vlc & 0x7u32) as usize] as OPJ_UINT32; // u value
820    vlc >>= d & 0x3u32; // kappa is 1 for initial line
821    consumed_bits =
822      (consumed_bits as core::ffi::c_uint).wrapping_add(d & 0x3u32) as OPJ_UINT32 as OPJ_UINT32;
823    suffix_len = d >> 2i32 & 0x7u32;
824    consumed_bits = (consumed_bits as core::ffi::c_uint).wrapping_add(suffix_len) as OPJ_UINT32;
825    d = (d >> 5i32).wrapping_add(vlc & ((1u32) << suffix_len).wrapping_sub(1u32));
826    *u.offset(0) = if mode == 1u32 {
827      d.wrapping_add(1u32)
828    } else {
829      1u32
830    };
831    *u.offset(1) = if mode == 1u32 {
832      1u32
833    } else {
834      d.wrapping_add(1u32)
835    }
836  } else if mode == 3u32 {
837    // both u_off are 1, and MEL event is 0
838    let mut d1 = dec[(vlc & 0x7u32) as usize] as OPJ_UINT32; // LSBs of VLC are prefix codeword
839    vlc >>= d1 & 0x3u32; // Consume bits
840    consumed_bits =
841      (consumed_bits as core::ffi::c_uint).wrapping_add(d1 & 0x3u32) as OPJ_UINT32 as OPJ_UINT32;
842    if d1 & 0x3u32 > 2u32 {
843      let mut suffix_len_0: OPJ_UINT32 = 0;
844      //Kappa is 1 for initial line
845      *u.offset(1) = (vlc & 1u32).wrapping_add(1u32).wrapping_add(1u32);
846      consumed_bits = consumed_bits.wrapping_add(1);
847      vlc >>= 1i32;
848      suffix_len_0 = d1 >> 2i32 & 0x7u32;
849      consumed_bits = (consumed_bits as core::ffi::c_uint).wrapping_add(suffix_len_0) as OPJ_UINT32;
850      d1 = (d1 >> 5i32).wrapping_add(vlc & ((1u32) << suffix_len_0).wrapping_sub(1u32));
851      *u.offset(0) = d1.wrapping_add(1u32)
852    } else {
853      let mut d2: OPJ_UINT32 = 0;
854      let mut suffix_len_1: OPJ_UINT32 = 0;
855      //u_{q_2} prefix
856      //Kappa is 1 for initial line
857      // u value
858      //Kappa is 1 for initial line
859      d2 = dec[(vlc & 0x7u32) as usize] as OPJ_UINT32; // LSBs of VLC are prefix codeword
860      vlc >>= d2 & 0x3u32; // Consume bits
861      consumed_bits =
862        (consumed_bits as core::ffi::c_uint).wrapping_add(d2 & 0x3u32) as OPJ_UINT32 as OPJ_UINT32; // u value
863      suffix_len_1 = d1 >> 2i32 & 0x7u32; //Kappa is 1 for initial line
864      consumed_bits = (consumed_bits as core::ffi::c_uint).wrapping_add(suffix_len_1) as OPJ_UINT32; // u value
865      d1 = (d1 >> 5i32).wrapping_add(vlc & ((1u32) << suffix_len_1).wrapping_sub(1u32));
866      *u.offset(0) = d1.wrapping_add(1u32);
867      vlc >>= suffix_len_1;
868      suffix_len_1 = d2 >> 2i32 & 0x7u32;
869      consumed_bits = (consumed_bits as core::ffi::c_uint).wrapping_add(suffix_len_1) as OPJ_UINT32;
870      d2 = (d2 >> 5i32).wrapping_add(vlc & ((1u32) << suffix_len_1).wrapping_sub(1u32));
871      *u.offset(1) = d2.wrapping_add(1u32)
872    }
873  } else if mode == 4u32 {
874    // both u_off are 1, and MEL event is 1
875    let mut d1_0: OPJ_UINT32 = 0; // LSBs of VLC are prefix codeword
876    let mut d2_0: OPJ_UINT32 = 0; // Consume bits
877    let mut suffix_len_2: OPJ_UINT32 = 0; // LSBs of VLC are prefix codeword
878    d1_0 = dec[(vlc & 0x7u32) as usize] as OPJ_UINT32; // Consume bits
879    vlc >>= d1_0 & 0x3u32; // u value
880    consumed_bits =
881      (consumed_bits as core::ffi::c_uint).wrapping_add(d1_0 & 0x3u32) as OPJ_UINT32 as OPJ_UINT32; // add 2+kappa
882    d2_0 = dec[(vlc & 0x7u32) as usize] as OPJ_UINT32; // u value
883    vlc >>= d2_0 & 0x3u32;
884    consumed_bits =
885      (consumed_bits as core::ffi::c_uint).wrapping_add(d2_0 & 0x3u32) as OPJ_UINT32 as OPJ_UINT32;
886    suffix_len_2 = d1_0 >> 2i32 & 0x7u32;
887    consumed_bits = (consumed_bits as core::ffi::c_uint).wrapping_add(suffix_len_2) as OPJ_UINT32;
888    d1_0 = (d1_0 >> 5i32).wrapping_add(vlc & ((1u32) << suffix_len_2).wrapping_sub(1u32));
889    *u.offset(0) = d1_0.wrapping_add(3u32);
890    vlc >>= suffix_len_2;
891    suffix_len_2 = d2_0 >> 2i32 & 0x7u32;
892    consumed_bits = (consumed_bits as core::ffi::c_uint).wrapping_add(suffix_len_2) as OPJ_UINT32;
893    d2_0 = (d2_0 >> 5i32).wrapping_add(vlc & ((1u32) << suffix_len_2).wrapping_sub(1u32));
894    *u.offset(1) = d2_0.wrapping_add(3u32)
895  }
896  consumed_bits
897}

982/* * @brief Read and unstuffs 32 bits from forward-growing bitstream
983 *
984 *  A subroutine to read from both the MagSgn or SPP bitstreams;
985 *  in particular, when MagSgn bitstream is consumed, 0xFF's are fed,
986 *  while when SPP is exhausted 0's are fed in.
987 *  X controls this value.
988 *
989 *  Unstuffing prevent sequences that are more than 0xFF7F from appearing
990 *  in the conpressed sequence.  So whenever a value of 0xFF is coded, the
991 *  MSB of the next byte is set 0 and must be ignored during decoding.
992 *
993 *  Reading can go beyond the end of buffer by up to 3 bytes.
994 *
995 *  @param  [in]  msp is a pointer to frwd_struct_t structure
996 *
997 */
998#[inline]
999unsafe fn frwd_read(mut msp: *mut frwd_struct_t) {
1000  let mut val: OPJ_UINT32 = 0; // assert that there is a space for 32 bits
1001  let mut bits: OPJ_UINT32 = 0; // read 32 bits
1002  let mut t: OPJ_UINT32 = 0;
1003  let mut unstuff: OPJ_BOOL = 0;
1004  assert!((*msp).bits <= 32u32);
1005  val = 0u32;
1006  if (*msp).size > 3i32 {
1007    val = read_le_uint32((*msp).data as *const core::ffi::c_void);
1008    // reduce size
1009    (*msp).data = (*msp).data.offset(4); // increment pointer
1010    (*msp).size -= 4i32
1011  } else if (*msp).size > 0i32 {
1012    let mut i = 0i32; // read one byte at a time
1013    val = if (*msp).X != 0u32 {
1014      0xffffffffu32
1015    } else {
1016      0u32
1017    }; // mask of location
1018    while (*msp).size > 0i32 {
1019      let fresh11 = (*msp).data; // put one byte in its correct location
1020      (*msp).data = (*msp).data.offset(1);
1021      let mut v = *fresh11 as OPJ_UINT32;
1022      let mut m = !((0xffu32) << i);
1023      val = val & m | v << i;
1024      (*msp).size -= 1;
1025      i += 8i32
1026    }
1027  } else {
1028    val = if (*msp).X != 0u32 {
1029      0xffffffffu32
1030    } else {
1031      0u32
1032    }
1033  }
1034  // we accumulate in t and keep a count of the number of bits in bits
1035  bits = (8u32).wrapping_sub(if (*msp).unstuff != 0 { 1u32 } else { 0u32 }); // Do we need unstuffing next?
1036  t = val & 0xffu32; // for next byte
1037  unstuff = (val & 0xffu32 == 0xffu32) as core::ffi::c_int; // move data to msp->tmp
1038  t |= (val >> 8i32 & 0xffu32) << bits;
1039  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(if unstuff != 0 {
1040    1u32
1041  } else {
1042    0u32
1043  })) as OPJ_UINT32;
1044  unstuff = (val >> 8i32 & 0xffu32 == 0xffu32) as core::ffi::c_int;
1045  t |= (val >> 16i32 & 0xffu32) << bits;
1046  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(if unstuff != 0 {
1047    1u32
1048  } else {
1049    0u32
1050  })) as OPJ_UINT32;
1051  unstuff = (val >> 16i32 & 0xffu32 == 0xffu32) as core::ffi::c_int;
1052  t |= (val >> 24i32 & 0xffu32) << bits;
1053  bits = (bits as core::ffi::c_uint).wrapping_add((8u32).wrapping_sub(if unstuff != 0 {
1054    1u32
1055  } else {
1056    0u32
1057  })) as OPJ_UINT32;
1058  (*msp).unstuff = (val >> 24i32 & 0xffu32 == 0xffu32) as core::ffi::c_int;
1059  (*msp).tmp |= (t as OPJ_UINT64) << (*msp).bits;
1060  (*msp).bits = ((*msp).bits as core::ffi::c_uint).wrapping_add(bits) as OPJ_UINT32;
1061}

1108      as OPJ_UINT32;
1109    (*msp).unstuff = (d & 0xffu64 == 0xffu64) as core::ffi::c_int;
1110    i += 1
1111  }

1471    /* The standard stipulates 2 <= Scup <= min(Lcup, 4079) */
1472    event_msg!(p_manager, EVT_ERROR,
1473                      "Malformed HT codeblock. One of the following condition is not met: 2 <= Scup <= min(Lcup, 4079)\n");

1475  }
1476  // init structures
1477  if !mel_init(&mut mel, coded_data, lcup, scup) {

1479      p_manager,
1480      EVT_ERROR,
1481      "Malformed HT codeblock.  Incorrect MEL segment sequence.\n"
1482    );

1484  }
1485  rev_init(&mut vlc, coded_data, lcup, scup);
1486  frwd_init(&mut magsgn, coded_data, lcup - scup, 0xff as OPJ_UINT32);

1490      &mut sigprop,
1491      coded_data.offset(lengths1 as isize),
1492      lengths2 as core::ffi::c_int,

1494    );

1498      &mut magref,
1499      coded_data,
1500      lengths1 as core::ffi::c_int,

1502    );

1504  /* * State storage
1505   *  One byte per quad; for 1024 columns, or 512 quads, we need
1506   *  512 bytes. We are using 2 extra bytes one on the left and one on
1507   *  the right for convenience.
1508   *
1509   *  The MSB bit in each byte is (\sigma^nw | \sigma^n), and the 7 LSBs
1510   *  contain max(E^nw | E^n)
1511   */

1513  // here expanded to 528

1515  //initial 2 lines
1516  // ///////////////

1518  *lsp.offset(0) = 0 as OPJ_UINT8; // for initial row of quad, we set to 0

1520                               // data represented as runs of 0 events
1521                               // See mel_decode description

1536    let mut locs: OPJ_UINT32 = 0;

1538    // ///////////
1539    //first quad
1540    // Get the head of the VLC bitstream. One fetch is enough for two
1541    // quads, since the largest VLC code is 7 bits, and maximum number of
1542    // bits used for u is 8.  Therefore for two quads we need 30 bits
1543    // (if we include unstuffing, then 32 bits are enough, since we have
1544    // a maximum of one stuffing per two bytes)

1571    //                               0 0 0 0 0 0 0 0
1572    //                               0 0 0 0 0 0 0 0

1574    //      nibble (4 bits) number   0 1 2 3 4 5 6 7
1575    //                         LSB   0 0 0 0 c c 0 0
1576    //                               0 0 0 0 c c 0 0

1596        }
1597      }

1604    // The update depends on the value of x; consider one OPJ_UINT32
1605    // if x is 0, 8, 16 and so on, then this line update c locations

1607    //                         LSB   0 0 c c 0 0 0 0
1608    //                               0 0 c c 0 0 0 0
1609    //                               0 0 0 0 0 0 0 0
1610    //                               0 0 0 0 0 0 0 0

1612    //      nibble (4 bits) number   0 1 2 3 4 5 6 7
1613    //                         LSB   0 0 0 0 0 0 c c

1648    // ///////////////////////////////////

1654    } // limits height

1656      locs

1658      (locs) & 0x55u32
1659    };

1681      // if this is inside the codeblock, set the
1682      *sp.offset(0) = 0 as OPJ_UINT32
1683      // sample to zero
1684    }
1685    if qinf[0_usize] & 0x20u32 != 0 {
1686      //sigma_n
1687      let mut val_0: OPJ_UINT32 = 0; //get 32 bits
1688      let mut t: OPJ_UINT32 = 0; //m_n, uses EMB e_k
1689      ms_val = frwd_fetch(&mut magsgn); //consume m_n
1690      m_n = U_q[0_usize].wrapping_sub(qinf[0_usize] >> 13i32 & 1u32); //get sign bit
1691      frwd_advance(&mut magsgn, m_n); //keep only m_n bits
1692      val_0 = ms_val << 31i32; //add EMB e_1
1693      v_n = ms_val & ((1u32) << m_n).wrapping_sub(1u32); //bin center
1694      v_n |= ((qinf[0_usize] & 0x200u32) >> 9i32) << m_n;
1695      v_n |= 1u32;
1696      //v_n now has 2 * (\mu - 1) + 0.5 with correct sign bit
1697      //add 2 to make it 2*\mu+0.5, shift it up to missing MSBs
1698      *sp.offset(stride as isize) = val_0 | v_n.wrapping_add(2u32) << p.wrapping_sub(1u32);
1699      //update line_state: bit 7 (\sigma^N), and E^N
1700      t = (*lsp.offset(0) as core::ffi::c_int & 0x7fi32) as OPJ_UINT32; // keep E^NW
1701      v_n = (32u32).wrapping_sub(count_leading_zeros(v_n));
1702      *lsp.offset(0) = (0x80u32 | (if t > v_n { t } else { v_n })) as OPJ_UINT8
1703    } else if locs & 0x2u32 != 0 {
1704      // if this is inside the codeblock, set the
1705      *sp.offset(stride as isize) = 0 as OPJ_UINT32
1706      // sample to zero
1707    } // move to next quad information
1708    lsp = lsp.offset(1); // move to next column of samples
1709    sp = sp.offset(1);
1710    //this is similar to the above two samples
1711    if qinf[0_usize] & 0x40u32 != 0 {

1722      *sp.offset(0) = 0 as OPJ_UINT32
1723    }
1724    *lsp.offset(0) = 0 as OPJ_UINT8;
1725    if qinf[0_usize] & 0x80u32 != 0 {
1726      let mut val_2: OPJ_UINT32 = 0;
1727      ms_val = frwd_fetch(&mut magsgn);
1728      m_n = U_q[0_usize].wrapping_sub(qinf[0_usize] >> 15i32 & 1u32);
1729      frwd_advance(&mut magsgn, m_n);
1730      val_2 = ms_val << 31i32;

1739      *sp.offset(stride as isize) = 0 as OPJ_UINT32
1740    } //move to next column
1741    sp = sp.offset(1);

1755    }

1772      *sp.offset(stride as isize) = 0 as OPJ_UINT32

1774      //E^NW
1775      //E^N
1776      //no need to update line_state
1777    } //move line state to next quad
1778    lsp = lsp.offset(1); //move to next sample

1792    }

1795      let mut val_6: OPJ_UINT32 = 0;

1844                                                                                         //the following is very similar to previous code, so please refer to
