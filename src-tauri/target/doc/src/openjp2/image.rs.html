66#[no_mangle]
67pub unsafe fn opj_image_create(
68  mut numcmpts: OPJ_UINT32,
69  mut cmptparms: *mut opj_image_cmptparm_t,
70  mut clrspc: OPJ_COLOR_SPACE,
71) -> *mut opj_image_t {
72  let mut compno: OPJ_UINT32 = 0;
73  let mut image = std::ptr::null_mut::<opj_image_t>();
74  image = opj_calloc(1i32 as size_t, core::mem::size_of::<opj_image_t>()) as *mut opj_image_t;
75  if !image.is_null() {
76    (*image).color_space = clrspc;
77    (*image).numcomps = numcmpts;
78    /* allocate memory for the per-component information */
79    (*image).comps = opj_calloc(
80      (*image).numcomps as size_t,
81      core::mem::size_of::<opj_image_comp_t>(),
82    ) as *mut opj_image_comp_t;
83    if (*image).comps.is_null() {
84      /* TODO replace with event manager, breaks API */
85      /* fprintf(stderr,"Unable to allocate memory for image.\n"); */
86      opj_image_destroy(image);
87      return std::ptr::null_mut::<opj_image_t>();
88    }
89    /* create the individual image components */
90    compno = 0 as OPJ_UINT32;
91    while compno < numcmpts {
92      let mut comp: *mut opj_image_comp_t =
93        &mut *(*image).comps.offset(compno as isize) as *mut opj_image_comp_t;
94      (*comp).dx = (*cmptparms.offset(compno as isize)).dx;
95      (*comp).dy = (*cmptparms.offset(compno as isize)).dy;
96      (*comp).w = (*cmptparms.offset(compno as isize)).w;
97      (*comp).h = (*cmptparms.offset(compno as isize)).h;
98      (*comp).x0 = (*cmptparms.offset(compno as isize)).x0;
99      (*comp).y0 = (*cmptparms.offset(compno as isize)).y0;
100      (*comp).prec = (*cmptparms.offset(compno as isize)).prec;
101      (*comp).sgnd = (*cmptparms.offset(compno as isize)).sgnd;
102      if (*comp).h != 0u32
103        && (*comp).w as OPJ_SIZE_T
104          > (usize::MAX)
105            .wrapping_div((*comp).h as usize)
106            .wrapping_div(core::mem::size_of::<OPJ_INT32>())
107      {
108        /* TODO event manager */
109        opj_image_destroy(image);
110        return std::ptr::null_mut::<opj_image_t>();
111      }
112      (*comp).data = opj_image_data_alloc(
113        ((*comp).w as size_t)
114          .wrapping_mul((*comp).h as usize)
115          .wrapping_mul(core::mem::size_of::<OPJ_INT32>()),
116      ) as *mut OPJ_INT32;
117      if (*comp).data.is_null() {
118        /* TODO replace with event manager, breaks API */
119        /* fprintf(stderr,"Unable to allocate memory for image.\n"); */
120        opj_image_destroy(image);
121        return std::ptr::null_mut::<opj_image_t>();
122      }
123      memset(
124        (*comp).data as *mut core::ffi::c_void,
125        0i32,
126        ((*comp).w as size_t)
127          .wrapping_mul((*comp).h as usize)
128          .wrapping_mul(core::mem::size_of::<OPJ_INT32>()),
129      );
130      compno += 1;
131    }
132  }
133  image
134}

136#[no_mangle]
137pub unsafe fn opj_image_destroy(mut image: *mut opj_image_t) {
138  if !image.is_null() {
139    if !(*image).comps.is_null() {
140      let mut compno: OPJ_UINT32 = 0;
141      /* image components */
142      compno = 0 as OPJ_UINT32;
143      while compno < (*image).numcomps {
144        let mut image_comp: *mut opj_image_comp_t =
145          &mut *(*image).comps.offset(compno as isize) as *mut opj_image_comp_t;
146        if !(*image_comp).data.is_null() {
147          opj_image_data_free((*image_comp).data as *mut core::ffi::c_void);
148        }
149        compno += 1;
150      }
151      opj_free((*image).comps as *mut core::ffi::c_void);
152    }
153    if !(*image).icc_profile_buf.is_null() {
154      opj_free((*image).icc_profile_buf as *mut core::ffi::c_void);
155    }
156    opj_free(image as *mut core::ffi::c_void);
157  };
158}

160/* *
161 * Updates the components characteristics of the image from the coding parameters.
162 *
163 * @param p_image_header    the image header to update.
164 * @param p_cp              the coding parameters from which to update the image.
165 */
166pub(crate) unsafe fn opj_image_comp_header_update(
167  mut p_image_header: *mut opj_image_t,
168  mut p_cp: *const opj_cp,
169) {
170  let mut i: OPJ_UINT32 = 0; /* validity of p_cp members used here checked in opj_j2k_read_siz. Can't overflow. */
171  let mut l_width: OPJ_UINT32 = 0; /* can't overflow */
172  let mut l_height: OPJ_UINT32 = 0; /* use add saturated to prevent overflow */
173  let mut l_x0: OPJ_UINT32 = 0; /* use add saturated to prevent overflow */
174  let mut l_y0: OPJ_UINT32 = 0;
175  let mut l_x1: OPJ_UINT32 = 0;
176  let mut l_y1: OPJ_UINT32 = 0;
177  let mut l_comp_x0: OPJ_UINT32 = 0;
178  let mut l_comp_y0: OPJ_UINT32 = 0;
179  let mut l_comp_x1: OPJ_UINT32 = 0;
180  let mut l_comp_y1: OPJ_UINT32 = 0;
181  let mut l_img_comp = std::ptr::null_mut::<opj_image_comp_t>();
182  l_x0 = opj_uint_max((*p_cp).tx0, (*p_image_header).x0);
183  l_y0 = opj_uint_max((*p_cp).ty0, (*p_image_header).y0);
184  l_x1 = (*p_cp)
185    .tx0
186    .wrapping_add((*p_cp).tw.wrapping_sub(1u32).wrapping_mul((*p_cp).tdx));
187  l_y1 = (*p_cp)
188    .ty0
189    .wrapping_add((*p_cp).th.wrapping_sub(1u32).wrapping_mul((*p_cp).tdy));
190  l_x1 = opj_uint_min(opj_uint_adds(l_x1, (*p_cp).tdx), (*p_image_header).x1);
191  l_y1 = opj_uint_min(opj_uint_adds(l_y1, (*p_cp).tdy), (*p_image_header).y1);
192  l_img_comp = (*p_image_header).comps;
193  i = 0 as OPJ_UINT32;
194  while i < (*p_image_header).numcomps {
195    l_comp_x0 = opj_uint_ceildiv(l_x0, (*l_img_comp).dx);
196    l_comp_y0 = opj_uint_ceildiv(l_y0, (*l_img_comp).dy);
197    l_comp_x1 = opj_uint_ceildiv(l_x1, (*l_img_comp).dx);
198    l_comp_y1 = opj_uint_ceildiv(l_y1, (*l_img_comp).dy);
199    l_width = opj_uint_ceildivpow2(l_comp_x1.wrapping_sub(l_comp_x0), (*l_img_comp).factor);
200    l_height = opj_uint_ceildivpow2(l_comp_y1.wrapping_sub(l_comp_y0), (*l_img_comp).factor);
201    (*l_img_comp).w = l_width;
202    (*l_img_comp).h = l_height;
203    (*l_img_comp).x0 = l_comp_x0;
204    (*l_img_comp).y0 = l_comp_y0;
205    l_img_comp = l_img_comp.offset(1);
206    i += 1;
207  }
208}

210/* *
211 * Copy only header of image and its component header (no data are copied)
212 * if dest image have data, they will be freed
213 *
214 * @param   p_image_src     the src image
215 * @param   p_image_dest    the dest image
216 *
217 */
218pub(crate) unsafe fn opj_copy_image_header(
219  mut p_image_src: *const opj_image_t,
220  mut p_image_dest: *mut opj_image_t,
221) {
222  let mut compno: OPJ_UINT32 = 0;
223  /* preconditions */
224
225  assert!(!p_image_src.is_null());
226  assert!(!p_image_dest.is_null());
227  (*p_image_dest).x0 = (*p_image_src).x0;
228  (*p_image_dest).y0 = (*p_image_src).y0;
229  (*p_image_dest).x1 = (*p_image_src).x1;
230  (*p_image_dest).y1 = (*p_image_src).y1;
231  if !(*p_image_dest).comps.is_null() {
232    compno = 0 as OPJ_UINT32;
233    while compno < (*p_image_dest).numcomps {
234      let mut image_comp: *mut opj_image_comp_t =
235        &mut *(*p_image_dest).comps.offset(compno as isize) as *mut opj_image_comp_t;
236      if !(*image_comp).data.is_null() {
237        opj_image_data_free((*image_comp).data as *mut core::ffi::c_void);
238      }
239      compno += 1;
240    }
241    opj_free((*p_image_dest).comps as *mut core::ffi::c_void);
242    (*p_image_dest).comps = std::ptr::null_mut::<opj_image_comp_t>()
243  }
244  (*p_image_dest).numcomps = (*p_image_src).numcomps;
245  (*p_image_dest).comps = opj_malloc(
246    ((*p_image_dest).numcomps as usize).wrapping_mul(core::mem::size_of::<opj_image_comp_t>()),
247  ) as *mut opj_image_comp_t;
248  if (*p_image_dest).comps.is_null() {
249    (*p_image_dest).comps = std::ptr::null_mut::<opj_image_comp_t>();
250    (*p_image_dest).numcomps = 0 as OPJ_UINT32;
251    return;
252  }
253  compno = 0 as OPJ_UINT32;
254  while compno < (*p_image_dest).numcomps {
255    memcpy(
256      &mut *(*p_image_dest).comps.offset(compno as isize) as *mut opj_image_comp_t
257        as *mut core::ffi::c_void,
258      &mut *(*p_image_src).comps.offset(compno as isize) as *mut opj_image_comp_t
259        as *const core::ffi::c_void,
260      core::mem::size_of::<opj_image_comp_t>(),
261    );
262    let fresh0 = &mut (*(*p_image_dest).comps.offset(compno as isize)).data;
263    *fresh0 = std::ptr::null_mut::<OPJ_INT32>();
264    compno += 1;
265  }
266  (*p_image_dest).color_space = (*p_image_src).color_space;
267  (*p_image_dest).icc_profile_len = (*p_image_src).icc_profile_len;
268  if (*p_image_dest).icc_profile_len != 0 {
269    (*p_image_dest).icc_profile_buf =
270      opj_malloc((*p_image_dest).icc_profile_len as size_t) as *mut OPJ_BYTE;
271    if (*p_image_dest).icc_profile_buf.is_null() {
272      (*p_image_dest).icc_profile_buf = std::ptr::null_mut::<OPJ_BYTE>();
273      (*p_image_dest).icc_profile_len = 0 as OPJ_UINT32;
274      return;
275    }
276    memcpy(
277      (*p_image_dest).icc_profile_buf as *mut core::ffi::c_void,
278      (*p_image_src).icc_profile_buf as *const core::ffi::c_void,
279      (*p_image_src).icc_profile_len as usize,
280    );
281  } else {
282    (*p_image_dest).icc_profile_buf = std::ptr::null_mut::<OPJ_BYTE>()
283  };
284}
