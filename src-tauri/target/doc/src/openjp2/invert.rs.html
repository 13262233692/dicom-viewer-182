8/* *
9 * Matrix inversion.
10 */
11#[no_mangle]
12pub(crate) unsafe fn opj_matrix_inversion_f(
13  mut pSrcMatrix: &mut [f32],
14  mut pDestMatrix: &mut [f32],
15  mut nb_compo: usize,
16) -> bool {
17  let mut lPermutations = SmallVec::<[u32; 0]>::with_capacity(nb_compo);
18  let mut src_tmp = SmallVec::<[f32; 0]>::with_capacity(nb_compo);
19  let mut dest_tmp = SmallVec::<[f32; 0]>::with_capacity(nb_compo);
20  let mut swap_area = SmallVec::<[f32; 0]>::with_capacity(nb_compo);
21  lPermutations.resize(nb_compo, 0);
22  src_tmp.resize(nb_compo, 0.0);
23  dest_tmp.resize(nb_compo, 0.0);
24  swap_area.resize(nb_compo, 0.0);
25  if !opj_lupDecompose(
26    pSrcMatrix,
27    lPermutations.as_mut_slice(),
28    swap_area.as_mut_slice(),
29    nb_compo,
30  ) {
31    return false;
32  }
33  opj_lupInvert(
34    pSrcMatrix,
35    pDestMatrix,
36    nb_compo,
37    lPermutations.as_slice(),
38    src_tmp.as_mut_slice(),
39    dest_tmp.as_mut_slice(),
40    swap_area.as_mut_slice(),
41  );
42  true
43}

74/* *
75 * LUP decomposition
76 */
77/*
78==========================================================
79   Local functions
80==========================================================
81*/
82fn opj_lupDecompose(
83  matrix: &mut [f32],
84  permutations: &mut [u32],
85  p_swap_area: &mut [f32],
86  nb_compo: usize,
87) -> bool {
88  let mut tmp_permutations = 0;
89  let mut k2 = 0;
90  let mut temp = 0.0;
91  let l_last_column = nb_compo - 1;
92  let l_swap_size = nb_compo * std::mem::size_of::<f32>();
93  let mut l_tmp_matrix = 0;
94  let mut l_column_matrix = 0;
95  let mut l_dest_matrix = 0;
96  let mut offset = 1;
97  let mut l_stride = nb_compo - 1;
98  // initialize permutations
99  for i in 0..nb_compo {
100    permutations[i] = i as u32;
101  }
102  // now make a pivot with column switch
103  for k in 0..l_last_column {
104    // take the middle element
105    l_column_matrix = l_tmp_matrix + k;
106    // make permutation with the biggest value in the column
107    let mut p = 0.0;
108    for i in k..nb_compo {
109      temp = matrix[l_column_matrix].abs();
110      if temp > p {
111        p = temp;
112        k2 = i;
113      }
114      // next line
115      l_column_matrix += nb_compo;
116    }
117    // a whole rest of 0 -> non singular
118    if p == 0.0 {
119      return false;
120    }
121    // should we permute ?
122    if k2 != k {
123      // exchange of line
124      // k2 > k
125      let dst_p = tmp_permutations + k2 - k;
126      // swap indices
127      permutations.swap(tmp_permutations, dst_p);
128      // and swap entire line.
129      l_column_matrix = l_tmp_matrix + (k2 - k) * nb_compo;
130      p_swap_area.copy_from_slice(&matrix[l_column_matrix..l_column_matrix + l_swap_size]);
131      matrix.copy_within(l_tmp_matrix..l_tmp_matrix + l_swap_size, l_column_matrix);
132      matrix[l_tmp_matrix..l_tmp_matrix + l_swap_size].copy_from_slice(&p_swap_area[..l_swap_size]);
133    }
134    // now update data in the rest of the line and line after
135    l_dest_matrix = l_tmp_matrix + k;
136    l_column_matrix = l_dest_matrix + nb_compo;
137    // take the middle element
138    let temp = matrix[l_dest_matrix];
139    l_dest_matrix += 1;
140    // now compute up data (i.e. coeff up of the diagonal).
141    for _ in offset..nb_compo {
142      // divide the lower column elements by the diagonal value
143      // matrix[i][k] /= matrix[k][k];
144      // p = matrix[i][k]
145      let p = matrix[l_column_matrix] / temp;
146      l_column_matrix += 1;
147      matrix[l_column_matrix] = p;
148
149      for j in offset..nb_compo {
150        matrix[l_dest_matrix + j] -= p * matrix[l_dest_matrix + j - 1];
151      }
152      // come back to the k+1th element
153      l_dest_matrix -= l_stride;
154      // go to kth element of the next line
155      l_column_matrix += k;
156    }
157    // offset is now k+2
158    offset += 1;
159    // 1 element less for stride
160    l_stride -= 1;
161    // next line
162    l_tmp_matrix += nb_compo;
163  }
164  true
165}
166/* *
167 * LUP solving
168 */
169fn opj_lupSolve(
170  p_result: &mut [f32],
171  p_matrix: &[f32],
172  p_vector: &[f32],
173  p_permutations: &[u32],
174  nb_compo: usize,
175  p_intermediate_data: &mut [f32],
176) {
177  let mut k: i32 = 0;
178  let mut i: usize = 0;
179  let mut j: usize = 0;
180  let mut sum: f32 = 0.;
181  let mut u: f32 = 0.;
182  let l_stride = nb_compo + 1;
183  let mut l_current_ptr: usize;
184  let mut l_intermediate_ptr: usize;
185  let mut l_dest_ptr: usize;
186  let mut l_tmp_matrix: usize;
187  let mut l_line_matrix = 0;
188  let mut l_begin_ptr = p_result.len() - 1;
189  let mut l_generated_data = 0;
190  let mut l_current_permutation_ptr = 0;
191  l_intermediate_ptr = 0;
192  l_generated_data = nb_compo - 1;
193  i = 0;
194  while i < nb_compo {
195    sum = 0.0;
196    l_current_ptr = 0;
197    l_tmp_matrix = l_line_matrix;
198    j = 1;
199    while j <= i {
200      sum += p_matrix[l_tmp_matrix] * p_intermediate_data[l_current_ptr];
201      l_tmp_matrix += 1;
202      l_current_ptr += 1;
203      j += 1;
204    }
205    p_intermediate_data[l_intermediate_ptr] =
206      p_vector[p_permutations[l_current_permutation_ptr] as usize] - sum;
207    l_line_matrix += nb_compo;
208    l_intermediate_ptr += 1;
209    l_current_permutation_ptr += 1;
210    i += 1;
211  }
212  l_line_matrix = p_matrix.len() - nb_compo - 1;
213  l_dest_ptr = nb_compo;
214  assert!(nb_compo != 0);
215  k = nb_compo as i32 - 1;
216  while k != -1 {
217    sum = 0.0;
218    l_tmp_matrix = l_line_matrix;
219    u = p_matrix[l_tmp_matrix];
220    l_line_matrix -= l_stride;
221    l_current_ptr = l_dest_ptr;
222    j = (k + 1) as usize;
223    while j < nb_compo {
224      sum += p_matrix[l_tmp_matrix] * p_result[l_current_ptr];
225      l_tmp_matrix += 1;
226      l_current_ptr += 1;
227      j += 1;
228    }
229    p_result[l_begin_ptr] = (p_intermediate_data[l_generated_data] - sum) / u;
230    l_generated_data -= 1;
231    l_begin_ptr -= 1;
232    l_dest_ptr -= 1;
233    k -= 1;
234  }
235}
236/* *
237 *LUP inversion (call with the result of lupDecompose)
238 */
239fn opj_lupInvert(
240  pSrcMatrix: &mut [f32],
241  pDestMatrix: &mut [f32],
242  nb_compo: usize,
243  pPermutations: &[u32],
244  p_src_temp: &mut [f32],
245  p_dest_temp: &mut [f32],
246  p_swap_area: &mut [f32],
247) {
248  let nb_compo_usize = nb_compo;
249  let mut lLineMatrix = pDestMatrix;
250  for j in 0..nb_compo {
251    let lCurrentPtr = &mut lLineMatrix[j + 1..];
252    p_src_temp.fill(0.0);
253    p_src_temp[j] = 1.0;
254    let p_src_temp_slice = &mut p_src_temp[..nb_compo_usize];
255    let p_dest_temp_slice = &mut p_dest_temp[..nb_compo_usize];
256    let pSrcMatrix_slice = &pSrcMatrix[..nb_compo_usize * nb_compo_usize];
257    let pPermutations_slice = &pPermutations[..nb_compo_usize];
258    let p_swap_area_slice = &mut p_swap_area[..nb_compo_usize];
259    opj_lupSolve(
260      p_dest_temp_slice,
261      pSrcMatrix_slice,
262      p_src_temp_slice,
263      pPermutations_slice,
264      nb_compo_usize,
265      p_swap_area_slice,
266    );
267    for i in 0..nb_compo_usize {
268      lCurrentPtr[i * nb_compo_usize] = p_dest_temp[i];
269    }
270  }
271}
